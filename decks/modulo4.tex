\begin{frame}[fragile]{Objetivos de la Sesión}


\begin{itemize}
  \item Analizar los marcos regulatorios internacionales y locales sobre Inteligencia Artificial (EU AI Act, Directivas MinTIC / Superintendencia).
  \item Determinar la responsabilidad ética, disciplinaria y civil del abogado en el uso de IA.
  \item Redactar e implementar la Política Institucional de Uso de IA en la Firma.
  \item Diseñar la hoja de ruta para la adopción tecnológica responsable.
\end{itemize}
\end{frame}



\begin{frame}[fragile]{El Marco Regulatorio de la IA}


\begin{itemize}
  \item \textbf{El Principio del Riesgo (EU AI Act):} Clasificación de sistemas según su nivel de riesgo (Inaceptable, Alto, Limitado, Mínimo). Los sistemas usados en la administración de justicia o decisiones legales se consideran de alto riesgo.
  \item \textbf{Normativa en Colombia:} Circular de la Superintendencia de Industria y Comercio sobre tratamiento de datos en IA y principios éticos del Gobierno Nacional.
\end{itemize}
\end{frame}



\begin{frame}[fragile]{Responsabilidad Profesional y Ética}


\begin{itemize}
  \item \textbf{El Principio ``Human-in-the-Loop'':} La decisión y revisión final de cualquier producto legal corresponde de forma indelegable al profesional del derecho.
  \item \textbf{Secreto Profesional y Confidencialidad:} Prohibición estricta de subir secretos comerciales o datos de clientes a modelos públicos sin Zero Data Retention (ZDR).
  \item \textbf{Deber de Diligencia:} La falta de verificación de citas o precedentes inexistentes (``hallucinated cases'') constituye una falta disciplinaria ante el Consejo Superior de la Judicatura.
\end{itemize}
\end{frame}



\begin{frame}[fragile]{Estructura de la Política Interna de la Firma}


1. \textbf{Herramientas Autorizadas:} Lista blanca (White-list) de software e IA aprobadas por la dirección IT/Compliance.
2. \textbf{Clasificación de Información:} Matriz que define qué datos (Públicos, Internos, Confidenciales, Estrictamente Secretos) pueden procesarse con qué herramientas.
3. \textbf{Disclosure a Clientes:} Transparencia sobre el uso de IA en la facturación y elaboración de entregables cuando aplique.
4. \textbf{Sanciones y Auditoría:} Mecanismos de control interno.

\end{frame}



\begin{frame}[fragile]{Hoja de Ruta para la Adopción Institucional}


\begin{itemize}
  \item \textbf{Fase 1:} Diagnóstico de procesos repetitivos y mapeo de riesgos.
  \item \textbf{Fase 2:} Capacitación técnica y ética de todo el equipo de abogados y paralegales.
  \item \textbf{Fase 3:} Implementación de proyectos piloto en áreas controladas (ej. investigación o revisión inicial).
  \item \textbf{Fase 4:} Auditoría continua y ajuste de políticas.
\end{itemize}
\end{frame}



\begin{frame}[fragile]{Cierre del Curso y Certificación}


\begin{itemize}
  \item \textbf{Evaluación Final:} Presentación de la propuesta de Política de IA adaptada a cada firma participante.
  \item \textbf{Conclusiones:} La IA no reemplaza al abogado; el abogado que usa IA reemplaza al que no la usa.
\end{itemize}
\end{frame}
