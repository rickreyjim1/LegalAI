\begin{frame}[fragile]{1. The Scenario: The Perfect Storm}


\textbf{The Context:}
You are managing the ``Horizon'' project. Yesterday, the client's Project Sponsor sent a furious email:
1. They are withholding the Phase 2 payment, claiming the team missed the \textit{Response Time SLA} during a minor outage.
2. They are demanding a new ``Automated Backup Dashboard'', claiming it was implicitly included in the original scope.

\textbf{Your Goal:} Protect the revenue, reject the scope creep, and de-escalate the anger.

\end{frame}



\begin{frame}[fragile]{Exercise 1: Step 1 - SLA Defense}


\textbf{Task:} Analyze the SOW to prove Capgemini is compliant.

\begin{itemize}
  \item \textbf{Provide Copilot with the SOW snippet:} \textit{(Assume the SOW states: ``Capgemini will respond to Sev-3 minor outages within 4 business hours.'')}
  \item \textbf{Draft Prompt to Try:}
\end{itemize}
\textit{``Act as a Contract Analyst. Review this SLA snippet: [Paste Snippet]. The client claims we missed the SLA because we took 3 hours to respond to a minor outage on a Saturday. Based strictly on the text, did Capgemini breach the SLA? Explain why using Chain-of-Thought reasoning.''}

\end{frame}



\begin{frame}[fragile]{Exercise 2: Step 2 - The Change Request}


\textbf{Task:} Draft the CR for the new dashboard using the CARE framework.

\begin{itemize}
  \item \textbf{Draft Prompt to Try:}
\end{itemize}
*"Draft the internal justification for a Change Request using the CARE framework.
\textbf{Context:} The SOW only covers manual backup scripts.
\textbf{Action:} Client is demanding an automated visual dashboard.
\textbf{Result:} It requires 40 hours of effort (\$4,000).
\textbf{Evaluation:} We can deliver it in Phase 3 if they sign the CR this week."*

\end{frame}



\begin{frame}[fragile]{Exercise 3: Step 3 - The Diplomatic Email}


\textbf{Task:} Combine the SLA defense and the CR into one email using the TAM framework.

\begin{itemize}
  \item \textbf{Draft Prompt to Try:}
\end{itemize}
*"Now, draft the email to the frustrated Project Sponsor.
\textbf{Task:} Politely inform them that the SLA was not breached (explain the business hours rule gently) and attach the drafted Change Request for the dashboard they requested.
\textbf{Audience:} A tense executive who feels ignored.
\textbf{Method:} Highly empathetic, professional, firm on the commercial boundaries. Use the 5 Pillars of prompting to structure your output."*

\end{frame}
